cmake_minimum_required(VERSION 3.25)
project(MygrainBridge VERSION 0.8.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set_source_files_properties(src/plugin.cpp PROPERTIES LANGUAGE OBJCXX)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VST3_SDK_ROOT "" CACHE PATH "Pinned Steinberg VST3 SDK checkout")
if(NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Set VST3_SDK_ROOT to the SDK checkout; see README.md")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
option(MYGRAIN_BUILD_VALIDATOR "Build Steinberg validator" OFF)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ${MYGRAIN_BUILD_VALIDATOR} CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk)
if(MINGW)
  # Unused SDK DataExchange helper assumes MSVC aligned_alloc; bridge does not use it.
  get_target_property(sdk_sources sdk SOURCES)
  list(FILTER sdk_sources EXCLUDE REGEX "utility/dataexchange\\.cpp$")
  set_property(TARGET sdk PROPERTY SOURCES "${sdk_sources}")
endif()
add_library(MygrainBridge MODULE src/plugin.cpp src/server.cpp "${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
target_include_directories(MygrainBridge PRIVATE "${VST3_SDK_ROOT}" src)
target_link_libraries(MygrainBridge PRIVATE sdk)
if(WIN32)
  enable_language(RC)
  target_include_directories(MygrainBridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_sources(MygrainBridge PRIVATE resources.rc)
  target_sources(MygrainBridge PRIVATE "${VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp" src/exports.def)
  target_compile_definitions(MygrainBridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(MygrainBridge PRIVATE ws2_32 bcrypt shell32 advapi32 user32 gdi32 gdiplus ole32)
  if(MINGW)
    target_compile_options(MygrainBridge PRIVATE -Wall -Wextra)
    target_link_options(MygrainBridge PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  set_target_properties(MygrainBridge PROPERTIES PREFIX "" SUFFIX ".vst3")
elseif(APPLE)
  target_link_libraries(MygrainBridge PRIVATE "-framework Cocoa" "-framework Security")
else()
  target_link_libraries(MygrainBridge PRIVATE pthread crypto)
endif()
add_executable(bridge-core-test tests/core_test.cpp)

if(MSVC)

  target_compile_options(bridge-core-test PRIVATE -UNDEBUG)

  target_link_libraries(bridge-fixture PRIVATE pthread crypto)

# Real macOS bundles: no OpenSSL/Homebrew runtime; universal arm64 + x86_64.
option(MYGRAIN_BUILD_CEF "Build the separate MyGrain CEF instrument" OFF)
if(MYGRAIN_BUILD_CEF)
  add_executable(cef-preview-test cef/preview_test.cpp)
  add_library(MyGrainCEF MODULE src/plugin.cpp src/server.cpp "${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
  target_include_directories(MyGrainCEF PRIVATE "${VST3_SDK_ROOT}" src)
  target_compile_definitions(MyGrainCEF PRIVATE MYGRAIN_CEF)
  target_link_libraries(MyGrainCEF PRIVATE sdk)
  if(WIN32)
    target_sources(MyGrainCEF PRIVATE "${VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp" src/exports.def)
    target_compile_definitions(MyGrainCEF PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(MyGrainCEF PRIVATE ws2_32 bcrypt shell32 advapi32 user32 gdi32 gdiplus ole32)
    if(MINGW)
      target_link_options(MyGrainCEF PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    set_target_properties(MyGrainCEF PROPERTIES PREFIX "" SUFFIX ".vst3")
    add_executable(cef-windows-host tests/windows_host.cpp "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
    target_include_directories(cef-windows-host PRIVATE "${VST3_SDK_ROOT}")
    target_compile_definitions(cef-windows-host PRIVATE MYGRAIN_CEF_HOST UNICODE _UNICODE NOMINMAX)
    target_link_libraries(cef-windows-host PRIVATE sdk sdk_hosting user32 gdi32)
    if(MINGW)
      target_link_options(cef-windows-host PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  elseif(APPLE)
    add_executable(cef-mac-host cef/mac_host.mm "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
    target_include_directories(cef-mac-host PRIVATE "${VST3_SDK_ROOT}")
    target_link_libraries(cef-mac-host PRIVATE sdk sdk_hosting "-framework Cocoa")
    set(MYGRAIN_BUNDLE_NAME "MyGrain CEF")
    set(MYGRAIN_BUNDLE_ID "com.blackcarburning.mygrain.cef")
    set(MYGRAIN_BUNDLE_VERSION "0.1.3")
    configure_file(mac/Info.plist.in "${CMAKE_CURRENT_BINARY_DIR}/MyGrainCEF-Info.plist" @ONLY)
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/MyGrainCEF-Info.plist" cef_plist)
    string(REPLACE "<string>11.0</string>" "<string>13.0</string>" cef_plist "${cef_plist}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/MyGrainCEF-Info.plist" "${cef_plist}")
    target_sources(MyGrainCEF PRIVATE "${VST3_SDK_ROOT}/public.sdk/source/main/macmain.cpp")
    target_link_libraries(MyGrainCEF PRIVATE "-framework Cocoa" "-framework Security")
    set_target_properties(MyGrainCEF PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION vst3 OUTPUT_NAME "MyGrain CEF" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/MyGrainCEF-Info.plist" CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  else()
    target_link_libraries(MyGrainCEF PRIVATE pthread crypto)
  endif()
endif()
if(APPLE)
  foreach(target MygrainBridge MygrainFX)
    if(target STREQUAL "MygrainFX")
      set(MYGRAIN_BUNDLE_NAME "MYGRAIN FX")
      set(MYGRAIN_BUNDLE_ID "com.blackcarburning.mygrain.fx")
      set(MYGRAIN_BUNDLE_VERSION "0.4.0")
    else()
      set(MYGRAIN_BUNDLE_NAME "MYGRAIN Bridge")
      set(MYGRAIN_BUNDLE_ID "com.blackcarburning.mygrain.bridge")
      set(MYGRAIN_BUNDLE_VERSION "0.8.0")
    endif()
    configure_file(mac/Info.plist.in "${CMAKE_CURRENT_BINARY_DIR}/${target}-Info.plist" @ONLY)
    target_sources(${target} PRIVATE "${VST3_SDK_ROOT}/public.sdk/source/main/macmain.cpp" assets/mygrain-logo.png)
    set_source_files_properties(assets/mygrain-logo.png PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    file(GLOB pot_art "${CMAKE_CURRENT_SOURCE_DIR}/assets/pot-*.png")
    target_sources(${target} PRIVATE ${pot_art})
    set_source_files_properties(${pot_art} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_link_libraries(${target} PRIVATE "-framework Cocoa" "-framework Security")
    set_target_properties(${target} PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION vst3
      OUTPUT_NAME "${MYGRAIN_BUNDLE_NAME}"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/${target}-Info.plist"
      CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  endforeach()

endif()
