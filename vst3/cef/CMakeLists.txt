cmake_minimum_required(VERSION 3.25)
project(MyGrainCEFHelper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CEF_ROOT "" CACHE PATH "Pinned CEF 156.0.28 minimal SDK")
if(NOT EXISTS "${CEF_ROOT}/include/cef_version.h")
 message(FATAL_ERROR "Set CEF_ROOT to the pinned minimal binary distribution")
endif()
if(APPLE)
 enable_language(OBJCXX)
 set(CMAKE_OBJCXX_STANDARD 20)
 set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
 set_source_files_properties(helper.cpp PROPERTIES LANGUAGE OBJCXX)
endif()
add_executable(MyGrainCEF helper.cpp)
target_include_directories(MyGrainCEF PRIVATE "${CEF_ROOT}")
target_compile_definitions(MyGrainCEF PRIVATE CEF_API_VERSION=15600)
if(WIN32)
 target_compile_definitions(MyGrainCEF PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
 target_link_libraries(MyGrainCEF PRIVATE "${CEF_ROOT}/Release/libcef.lib" shell32 ole32 crypt32 uuid)
 if(MINGW)
  target_link_options(MyGrainCEF PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
elseif(APPLE)
 target_link_libraries(MyGrainCEF PRIVATE "${CEF_ROOT}/Release/Chromium Embedded Framework.framework/Chromium Embedded Framework" "-framework Cocoa")
 set_target_properties(MyGrainCEF PROPERTIES BUILD_RPATH "@executable_path/../Frameworks;@executable_path/../../../" INSTALL_RPATH "@executable_path/../Frameworks;@executable_path/../../../")
else()
 target_link_libraries(MyGrainCEF PRIVATE "${CEF_ROOT}/Release/libcef.so" pthread)
 set_target_properties(MyGrainCEF PROPERTIES BUILD_RPATH "${CEF_ROOT}/Release" INSTALL_RPATH "$ORIGIN")
endif()
